CREATE DATABASE UMS;
USE UMS;
-- 1. Department Table
CREATE TABLE Department (
    department_id VARCHAR(10) PRIMARY KEY,
    dept_nam VARCHAR(100) NOT NULL
);

-- 2. Staff Table (Relationship: Department employs Staff)
CREATE TABLE Staff (
    staff_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    dept_id VARCHAR(10) NOT NULL,
    FOREIGN KEY (dept_id) REFERENCES Department(department_id)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- 3. Professor Table (Relationship: Department employs Professor)
CREATE TABLE Professor (
    prof_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    designation VARCHAR(50),
    email VARCHAR(100) UNIQUE NOT NULL,
    dept_id VARCHAR(10) NOT NULL,
    FOREIGN KEY (dept_id) REFERENCES Department(department_id)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- 4. Student Table (Relationship: Department admits Student)
CREATE TABLE Student (
    student_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    DOB DATE,
    enrollment_date DATE NOT NULL,
    dept_id VARCHAR(10) NOT NULL,
    FOREIGN KEY (dept_id) REFERENCES Department(department_id)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- 5. Parents Table (Relationship: Student has Parents/Guardians)
CREATE TABLE Parents (
    student_id VARCHAR(10) NOT NULL,
    parent_name VARCHAR(100) NOT NULL,
    address VARCHAR(255),
    phone_number VARCHAR(20),
    PRIMARY KEY (student_id, parent_name),
    FOREIGN KEY (student_id) REFERENCES Student(student_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

-- 6. Course Table (Relationship: Professor teaches Course)
CREATE TABLE Course (
    course_id VARCHAR(10) PRIMARY KEY,
    course_title VARCHAR(100) NOT NULL,
    credit INT CHECK (credit > 0),
    prof_id VARCHAR(10),
    FOREIGN KEY (prof_id) REFERENCES Professor(prof_id)
        ON DELETE SET NULL ON UPDATE CASCADE
);

-- 7. Enrollment Table (Relationship: Student, Course, Department)
CREATE TABLE Enrollment (
    enrollment_id VARCHAR(10) PRIMARY KEY,
    student_id VARCHAR(10) NOT NULL,
    course_id VARCHAR(10) NOT NULL,
    dept_id VARCHAR(10) NOT NULL,
    semester VARCHAR(20) NOT NULL,
    FOREIGN KEY (student_id) REFERENCES Student(student_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (course_id) REFERENCES Course(course_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (dept_id) REFERENCES Department(department_id)
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT unique_student_course_semester UNIQUE (student_id, course_id, semester)
);

ADD COLUMN marks DECIMAL(5, 2) CHECK (marks >= 0 AND marks <= 100);
